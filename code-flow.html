@Service

    @Transactional
    public Order createOrder(...) {

        double subtotal = cart.getSubtotal();
        double discount = (promoCode != null) ? applyPromotion(cart, promoCode) : 0;
        double tax = (subtotal - discount) * 0.10;
        double total = subtotal - discount + tax;

        // 3. Build & Save Order
        Order order = Order.builder()...build();
        order = orderRepository.save(order);

        // 4. Clear Cart
        cartItemRepository.deleteAllByCartId(cart.getId());
        cartRepository.delete(cart);

        return order;
    }

// VnPayService.java
public String createPaymentUrl(Payment payment) {
    // 1. Build parameters (sorted alphabetically)
    Map<String, String> params = new TreeMap<>();
    params.put("vnp_Version", "2.1.0");
    params.put("vnp_Command", "pay");
    params.put("vnp_TmnCode", tmnCode);
    params.put("vnp_Amount", String.valueOf(payment.getAmount() * 100));
    params.put("vnp_TxnRef", payment.getTransactionId());
    params.put("vnp_ReturnUrl", returnUrl);
    params.put("vnp_CreateDate", now.format("yyyyMMddHHmmss"));

    // 2. Create query string
    String query = params.entrySet().stream()
        .map(e -> e.getKey() + "=" + URLEncode(e.getValue()))
        .collect(joining("&"));

    // 3. Generate HMAC-SHA512 signature
    String signature = hmacSHA512(hashSecret, query);

    // 4. Build final URL
    return apiUrl + "?" + query + "&vnp_SecureHash=" + signature;
}

VNPay Callback Verification

// VnPayService.verifyCallback()
public Payment verifyCallback(String callbackData) {
    // 1. Parse query string to map
    Map<String, String> params = parseQueryString(callbackData);

    // 2. Extract secure hash and order info
    String secureHash = params.remove("vnp_SecureHash");
    String orderInfo = params.get("vnp_OrderInfo");

    // 3. Recreate signature from params
    String signData = rebuildQueryString(params);
    String calculatedHash = hmacSHA512(hashSecret, signData);

    // 4. Compare signatures
    if (!secureHash.equals(calculatedHash)) {
        // Signature mismatch - potential fraud
        return payment.setStatus("failed");
    }

    // 5. Check response code
    String responseCode = params.get("vnp_ResponseCode");
    if ("00".equals(responseCode)) {
        // Payment successful
        payment.setStatus("completed");
        payment.setPaidAt(Instant.now());
    } else {
        payment.setStatus("failed");
    }

    return payment;
}
